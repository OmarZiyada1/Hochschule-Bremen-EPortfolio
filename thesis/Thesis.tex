\newcommand{\abbildung}[5]{%

		\centering
		\IfFileExists{bilder/#1}{%
			\includegraphics[width=#2\textwidth,height=0.62\textheight,keepaspectratio]{#1}%
		}{%
			\fcolorbox{gray!50}{gray!08}{%
				\parbox[c][4.2cm][c]{\dimexpr#2\textwidth-2\fboxsep-2\fboxrule\relax}{%
					\centering\small\itshape Abbildung noch nicht abgelegt\\[0.5em]
					\normalfont\small\texttt{bilder/#1}}}%
		}
		\caption[#4]{#5}
		\label{#3}
	\end{figure}}
